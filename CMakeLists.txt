cmake_minimum_required(VERSION 3.21)

project(Anim VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# GLAD is an older vendored project; keep its old policy baseline compatible
# with current CMake without changing the vendored source.
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
if(POLICY CMP0141)
  cmake_policy(SET CMP0141 NEW)
endif()

foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  string(TOUPPER "${config}" config_upper)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config_upper} "${CMAKE_BINARY_DIR}/bin")
  set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${config_upper} "${CMAKE_BINARY_DIR}/bin")
  set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${config_upper} "${CMAKE_BINARY_DIR}/lib")
endforeach()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")

file(COPY resources DESTINATION "${CMAKE_BINARY_DIR}/bin")
file(COPY imgui.ini LICENSE DESTINATION "${CMAKE_BINARY_DIR}/bin")
file(COPY mp_mocap DESTINATION "${CMAKE_BINARY_DIR}/bin")

# Use an installed Python for embedding. Configuring must not download or
# compile a second interpreter; scripts/setup.ps1 prepares the local venv.
if(WIN32)
  set(_anim_python_default "${CMAKE_BINARY_DIR}/bin/python/Scripts/python.exe")
else()
  set(_anim_python_default "${CMAKE_BINARY_DIR}/bin/python/bin/python")
endif()
if(EXISTS "${_anim_python_default}" AND NOT Python3_EXECUTABLE)
  set(Python3_EXECUTABLE "${_anim_python_default}" CACHE FILEPATH "Python used by Anim")
endif()
find_package(Python3 3.10 EXACT REQUIRED COMPONENTS Interpreter Development)
message(STATUS "Python runtime: ${Python3_EXECUTABLE}")
message(STATUS "Python headers: ${Python3_INCLUDE_DIRS}")
message(STATUS "Python library: ${Python3_LIBRARIES}")
if(WIN32)
  get_filename_component(_anim_python_lib_dir "${Python3_LIBRARIES}" DIRECTORY)
  get_filename_component(_anim_python_home "${_anim_python_lib_dir}" DIRECTORY)
  foreach(_python_dll python310.dll python3.dll)
    if(EXISTS "${_anim_python_home}/${_python_dll}")
      file(COPY "${_anim_python_home}/${_python_dll}" DESTINATION "${CMAKE_BINARY_DIR}/bin")
    endif()
  endforeach()
endif()

option(JSONCPP_WITH_TESTS "Build jsoncpp tests" OFF)
option(GLFW_BUILD_DOCS "Build GLFW documentation" OFF)
option(GLFW_BUILD_TESTS "Build GLFW tests" OFF)
option(GLFW_BUILD_EXAMPLES "Build GLFW examples" OFF)
option(ASSIMP_BUILD_ZLIB "Build Assimp's bundled zlib" ON)
option(ASSIMP_BUILD_TESTS "Build Assimp tests" OFF)
option(ASSIMP_BUILD_ASSIMP_TOOLS "Build Assimp tools" OFF)
option(ASSIMP_BUILD_SAMPLES "Build Assimp samples" OFF)
option(ASSIMP_BUILD_DOCS "Build Assimp documentation" OFF)
option(ASSIMP_INSTALL_PDB "Install Assimp debug symbol files" OFF)
option(ENABLE_TESTING "Build dependency tests" OFF)
set(ASSIMP_INSTALL_PDB OFF CACHE BOOL "Disable Assimp debug symbol files" FORCE)
set(ASSIMP_BUILD_ZLIB ON CACHE BOOL "Build Assimp's bundled zlib" FORCE)
set(ASSIMP_BUILD_ALL_IMPORTERS_BY_DEFAULT OFF CACHE BOOL "Build only required Assimp importers" FORCE)
set(ASSIMP_BUILD_FBX_IMPORTER ON CACHE BOOL "Import FBX models" FORCE)
set(ASSIMP_BUILD_GLTF_IMPORTER ON CACHE BOOL "Import glTF and VRM models" FORCE)
set(ASSIMP_BUILD_ALL_EXPORTERS_BY_DEFAULT OFF CACHE BOOL "Build only required Assimp exporters" FORCE)
set(ASSIMP_BUILD_FBX_EXPORTER ON CACHE BOOL "Export FBX models" FORCE)
set(ASSIMP_BUILD_GLTF_EXPORTER ON CACHE BOOL "Export glTF models" FORCE)

add_subdirectory(external/glad)
add_library(STB_IMAGE external/stb/stb_image.cpp)
add_subdirectory(external/glfw)
add_subdirectory(external/imgui)
add_subdirectory(external/assimp)
add_subdirectory(external/hiredis)
add_subdirectory(external/jsoncpp)
add_subdirectory(external/glm)
add_subdirectory(external/pybind11)

add_subdirectory(src)
add_subdirectory(playground/export_test)
